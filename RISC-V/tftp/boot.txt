# Automated Provisioning Script
echo "=== Starting Network Automation Script ==="
setenv autoload no
dhcp

# !!! THE CRUCIAL HARDWARE REALIGNMENT LAYER !!!
# Force a raw physical memory block copy from the internal pointer into RAM
fdt move $fdtcontroladdr 0x8c000000

echo "=== Uploading Device Tree Blob ==="
# Stream the physical bytes straight down the wire from our staging sector
tftpput 0x8c000000 0x19000 ${serverip}:${ethaddr}.dtb

echo "=== Pausing for I/O Processing ==="
sleep 5

echo "=== Downloading Application Binary via Port 80 ==="
wget $kernel_addr_r ${serverip}/kernel.bin

echo "=== Booting Target Program Natively ==="
go $kernel_addr_r
